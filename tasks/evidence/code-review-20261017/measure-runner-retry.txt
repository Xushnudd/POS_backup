// Run from the repository root with the isolated pos_test DATABASE_URL.
// Actual harness execution is required. Existing historical proof files stay unchanged.
const fs=require('node:fs'),{spawnSync,execFileSync}=require('node:child_process'),{createHash}=require('node:crypto');
const {sourceHashes,validateServiceTargets}=require('./scripts/release/check-service-targets.cjs');
const sha=bytes=>createHash('sha256').update(bytes).digest('hex'),prefix='tasks/evidence/';
const fixture=JSON.parse(fs.readFileSync('/tmp/pos-volume-dataset.json','utf8'));
const before=sourceHashes(process.cwd()),sourceRevision=execFileSync('git',['rev-parse','HEAD'],{encoding:'utf8'}).trim();
const startedAt=new Date().toISOString();
const run=spawnSync(process.execPath,['--require','./scripts/test-resolver.cjs','.test-build/tests/performance/core-measure.js'],{env:{...process.env,POS_PROFILE_TEST_ORG:fixture.organizationId,POS_PROFILE_PRODUCT_OFFSET:'30000'},encoding:'utf8',timeout:180000,maxBuffer:16*1024*1024});
const endedAt=new Date().toISOString(),after=sourceHashes(process.cwd()),unchanged=JSON.stringify(before)===JSON.stringify(after);
fs.writeFileSync('tasks/evidence/code-review-20261017/measure.log',(run.stdout||'')+(run.stderr||''),{flag:'wx'});
const execution={startedAt,endedAt,processExit:run.status,signal:run.signal,sourceRevision,sourceFilesUnchangedDuringMeasurement:unchanged,sourceFiles:Object.keys(before).length,sourceHashes:before,compiledHarnessSha256:sha(fs.readFileSync('.test-build/tests/performance/core-measure.js')),command:'node --require ./scripts/test-resolver.cjs .test-build/tests/performance/core-measure.js',environment:{POS_PROFILE_PRODUCT_OFFSET:30000},scope:'isolated_synthetic_service_measurement_not_deployment'};
const save=(name,data)=>{const path=prefix+name,bytes=Buffer.from(JSON.stringify(data,null,2)+'\n');fs.writeFileSync(path,bytes,{flag:'wx'});return {path,sha256:sha(bytes)};};
save('service-load-execution-20261017-code-review.json',execution);
if(run.status!==0||!unchanged)throw Error('Actual load run failed; no acceptance manifest published');
const load=JSON.parse(fs.readFileSync('/tmp/pos-volume-measure.json','utf8'));
if(Date.parse(load.at)<Date.parse(startedAt)||Date.parse(load.at)>Date.parse(endedAt)||load.fixtureOrganizationId!==fixture.organizationId)throw Error('Measurement identity/time mismatch');
const historical='tasks/evidence/release-service-targets-20261016-registration.json',old=JSON.parse(fs.readFileSync(historical,'utf8'));
const report={...old,sourceRevision,measuredAt:load.at,sourceHashes:before,load:save('service-load-20261017-code-review.json',load),fixture:save('service-load-dataset-20261017-code-review.json',fixture),execution:{path:prefix+'service-load-execution-20261017-code-review.json',sha256:sha(fs.readFileSync(prefix+'service-load-execution-20261017-code-review.json'))},documentationSha256:sha(fs.readFileSync('docs/operations/release-service-targets.md')),historical:[...old.historical,{path:historical,sha256:sha(fs.readFileSync(historical))}]};
for(const path of ['tasks/evidence/service-load-20261017-code-review-failed-1.json','tasks/evidence/service-load-execution-20261017-code-review-failed-1.json'])report.historical.push({path,sha256:sha(fs.readFileSync(path))});
const validated=validateServiceTargets(report);
save('release-service-targets-20261017-code-review.json',report);
console.log(JSON.stringify(validated));
